SELECT
  "ID_Client" AS value,
  "Label_Deroulant" AS label
FROM public."Clients"
WHERE "est_archive" = false
  AND (
    "ID_Client" = {{ appsmith.store.nouveau_client_id || '' }}
    OR (
      {{ !sel_client.filterText }} 
      OR "Label_Deroulant" ILIKE {{ '%' + sel_client.filterText + '%' }}
    )
  )
ORDER BY
  CASE WHEN "ID_Client" = {{ appsmith.store.nouveau_client_id || '' }} THEN 0 ELSE 1 END,
  "Label_Deroulant" ASC
LIMIT 50;
