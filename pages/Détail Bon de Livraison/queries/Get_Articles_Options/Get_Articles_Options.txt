SELECT
  "ID_Article" AS value,
  TRIM(
    COALESCE("Recherche_Article", COALESCE("Ref_Fournisseur", '') || ' - ' || COALESCE("Designation", '')) ||
    CASE 
      WHEN "Anciennes_References" IS NOT NULL AND TRIM("Anciennes_References") <> '' 
      THEN ' (Anc: ' || "Anciennes_References" || ')' 
      ELSE '' 
    END
  ) AS label,
  "Code_Barre",
  "Ref_Fournisseur",
  "Prix_Vente_TTC",
  "Anciennes_References"
FROM public."ARTICLES"
WHERE "Statut_Ligne" = 'Actif'
ORDER BY
  CASE WHEN "ID_Article" = {{ appsmith.store.nouvel_article_id || '' }} THEN 0 ELSE 1 END,
  "Recherche_Article" ASC;
