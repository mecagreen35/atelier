-- 2. Recalcul sur la ligne de commande + suppression de l'alerte
UPDATE public."LIGNES_COMMANDES_CLIENTS"
SET 
  "Prix_Vente_HT" = (
    SELECT "Prix_Vente_HT" 
    FROM public."ARTICLES" 
    WHERE "ID_Article" = '{{ Table_Lignes_Commande.updatedRow.ID_Article }}'
  ),
  "Prix_Achat" = {{ Number(Table_Lignes_Commande.updatedRow.Prix_Achat || 0) }},
  "Alerte_Prix" = false
WHERE "ID_Ligne_Client" = '{{ Table_Lignes_Commande.updatedRow.ID_Ligne_Client }}';
